```
# 1. Initialize a repository (once)
git init

# 2. Check status
git status

# 3. Add files
git add .

# 4. Commit changes
git commit -m "Initial commit"

# 5. Connect to a remote repository (once)
git remote add origin <repository-url>

# 6. Push to GitHub (first push)
git branch -M main
git push -u origin main

# 7. Future workflow
git add .
git commit -m "Describe your changes"
git push

```
